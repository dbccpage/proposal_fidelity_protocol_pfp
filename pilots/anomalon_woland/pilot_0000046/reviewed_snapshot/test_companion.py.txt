def request(graph="0>1<>0>2<>1>2", *, kind="complement", cutoff=2, degree=1, x=None):
    return {"task_id": "test", "kind": kind, "graph": graph, "cutoff": cutoff,
            "degree": degree, "cochain": ["1", "0", "0"] if x is None else x}


def graph_for(n, weighted=False):
    if n == 0:
        return "0"
    return "<>".join(f"{a}>{b}" + (f"w{1 + (3 * a + b) % 9}" if weighted else "")
                      for a, b in combinations(range(n + 1), 2))


def assert_rejected(req, module=M):
    with pytest.raises(ValueError):
        module.answer_for(req)


def test_dependency_pin_and_known_unweighted_triangle():
    core = M.get_core()
    assert M.sha(Path(core.__file__).read_bytes()) == M.CORE_SHA256
    for cutoff, harmonic, coexact in ((2, ["0"] * 3, ["2/3", "-2/3", "2/3"]),
                                     (1, ["2/3", "-2/3", "2/3"], ["0"] * 3)):
        result = M.answer_for(request(kind="unweighted", cutoff=cutoff, x=["1", "2", "3"]))
        data = result["data"]["data"]
        assert data["exact"] == ["1/3", "8/3", "7/3"]
        assert data["harmonic"] == harmonic and data["coexact"] == coexact
        assert data["boundary"]["columns"] == [["-1", "1", "0"], ["-1", "0", "1"], ["0", "-1", "1"]]
        assert data["boundary_squared_zero"] is True
        assert data["orthogonality"] == ["0", "0", "0"]
        assert data["harmonic_residual"] == ["0", "0", "0"]
    filled = M.answer_for(request(kind="unweighted"))["data"]["data"]
    assert filled["laplacian"] == [["3", "0", "0"], ["0", "3", "0"], ["0", "0", "3"]]
    assert filled["next_boundary"]["columns"] == [["1", "-1", "1"]]


def test_weights_and_direction_do_not_change_unweighted_basis():
    plain = M.answer_for(request(kind="unweighted"))["data"]
    reversed_edges = M.answer_for(request("2<1w3<>2<0q9<>1<0a4", kind="unweighted"))
    assert reversed_edges["data"] == plain
    assert reversed_edges["graph"]["vertices"] == ["0", "1", "2"]
    assert reversed_edges["graph"]["spellings"][0] == {"token": "2<1w3", "source": "1", "target": "2", "weight_marker": "w", "weight": 3}
    forward = M.answer_for(request("0>1w4<>0>2w9<>1>2w3"))
    backward = M.answer_for(request("1<0w4<>2<0w9<>2<1w3"))
    assert forward["data"] == backward["data"]


def test_parser_isolated_and_bounds():
    graph = M.parse_graph("z<>a<>0", 0)
    assert graph["vertices"] == ["0", "a", "z"]
    assert graph["weighted_edges"] == []
    assert graph["spellings"][0] == {"token": "z", "isolated": "z"}
    graph = M.parse_graph("0>1<>1>2<>2>3<>3>4<>4>5", 4)
    assert len(graph["vertices"]) == 6
    result = M.answer_for(request(graph_for(5), kind="unweighted", cutoff=4, degree=4, x=["0"] * 6))
    assert result["data"]["complex"]["truncated"] is True
    assert result["data"]["complex"]["clique_dimension"] == 5
    assert_rejected(request(graph_for(5), cutoff=4, degree=4, x=["0"] * 6))

def independent_orientation(face, other):
    left = list(face + other)
    inversions = 0
    for i in range(len(left)):
        for j in range(len(left) - 1 - i):
            if left[j] > left[j + 1]:
                left[j], left[j + 1] = left[j + 1], left[j]
                inversions += 1
    return -1 if inversions % 2 else 1


def test_all_62_augmented_basis_faces_unit_and_weighted():
    for weighted in (False, True):
        tested = 0
        for n in range(5):
            verts = tuple(str(i) for i in range(n + 1))
            edges = {edge: (1 + (3 * int(edge[0]) + int(edge[1])) % 9 if weighted else 1)
                     for edge in combinations(verts, 2)}
            for k in range(-1, n + 1):
                basis = list(combinations(verts, k + 1))
                for index, face in enumerate(basis):
                    x = [str(int(i == index)) for i in range(len(basis))]
                    data = M.answer_for(request(graph_for(n, weighted), cutoff=n, degree=k, x=x))["data"]
                    other = tuple(v for v in verts if v not in face)
                    destination = data["target_basis"].index(list(other))
                    value = data["value"][destination]
                    q, d = Fraction(value["coefficient"]), value["radicand"]
                    weight = lambda cell: math.prod(edges[edge] for edge in combinations(cell, 2))
                    assert q * q * d == Fraction(weight(other), weight(face))
                    assert (1 if q > 0 else -1) == independent_orientation(face, other)
                    assert all(row["coefficient"] == "0" for i, row in enumerate(data["value"]) if i != destination)
                    assert all(d % (p * p) for p in range(2, math.isqrt(d) + 1))
                    sign = -1 if (k + 1) * (n - k) % 2 else 1
                    assert data["twice"] == [{"coefficient": str(sign * int(v)), "radicand": 1} for v in x]
                    assert data["target_degree"] == n - k - 1 and data["double_sign"] == sign
                    tested += 1
        assert tested == 62

def test_byte_limits_and_square_guard(monkeypatch):
    task = request()
    task["oversized"] = ["x" * 2048] * 80
    assert_rejected(task)
    with pytest.raises(ValueError, match="byte limit"):
        M.evaluate({"oversized": ["x" * 2048] * 80})
    original = M.apply_complement
    calls = 0
    def corrupt_second(*args):
        nonlocal calls
        calls += 1
        degree, basis, values = original(*args)
        return degree, basis, tuple((q + 1, d) for q, d in values) if calls == 2 else values
    monkeypatch.setattr(M, "apply_complement", corrupt_second)
    with pytest.raises(ValueError, match="square check"):
        M.answer_for(request())


def test_task_binding_full_comparison_type_and_alias_custody():
    task = request()
    answer = M.answer_for(task)
    document = M.submission(task, answer)
    assert M.evaluate(document)["verdict"] == "MATCH"
    for key, replacement in (("task_id", "different"), ("graph", "1<0<>2<0<>2<1"), ("cochain", ["0", "1", "0"])):
        altered = copy.deepcopy(document)
        altered["request"][key] = replacement
        assert M.evaluate(altered)["verdict"] == "MISMATCH"
    changed = copy.deepcopy(document)
    changed["proposal"]["answer"]["data"]["double_sign"] = True
    assert M.evaluate(changed)["verdict"] == "MISMATCH"
    for value in (None, True, "0" * 64):
        changed = copy.deepcopy(document)
        changed["proposal"]["request_sha256"] = value
        assert M.evaluate(changed)["verdict"] == "MISMATCH"
    result = M.evaluate(document)
    document["request"]["cochain"][0] = "99"
    document["proposal"]["answer"]["data"]["value"][0]["coefficient"] = "99"
    assert result["request"]["cochain"][0] == "1"
    assert result["proposal"]["answer"]["data"]["value"][0]["coefficient"] == "0"
    assert answer["data"]["value"][0]["coefficient"] == "0"
    for value in ({}, {"request": request(), "proposal": {}}, {"request": request(), "proposal": {}, "extra": 1}):
        with pytest.raises(ValueError):
            M.evaluate(value)


def check_bundle(folder, raw, stdout):
    receipt = json.loads((folder / "receipt.json").read_bytes())
    assert receipt["input_complete"] is True and receipt["input_bytes"] == len(raw)
    assert (folder / "input.bin").read_bytes() == raw
    assert (folder / "stdout.txt").read_bytes() == stdout
    assert (folder / "stderr.txt").read_bytes() == b""
    for name, digest in receipt["artifact_sha256"].items():
        assert hashlib.sha256((folder / name).read_bytes()).hexdigest() == digest
    assert receipt["code_sha256"] == hashlib.sha256(RUNNER.read_bytes()).hexdigest()
    assert receipt["core_sha256"] == M.CORE_SHA256
    assert receipt["model_calls"] == 0 and receipt["cost_usd"] == "0"
    return json.loads((folder / "result.json").read_bytes())


@pytest.mark.parametrize("raw", [b"", b"{", b"\xff", b'{"x":NaN}', b'{"x":1,"x":2}', b'{"x":"\\ud800"}',
                                b"{" + b" " * (M.MAX_BYTES + 1), b"[]", b"[" * 2000 + b"]" * 2000],
                         ids=["empty", "syntax", "non_utf8", "nan", "duplicate_key", "surrogate", "oversized", "wrong_shape", "deep"])
def test_malformed_bytes_are_fully_retained(tmp_path, raw):
    source = tmp_path / "input.json"
    source.write_bytes(raw)
    status, stdout = M.run_file(source, tmp_path / "results")
    assert status == "REJECTED"
    result = check_bundle(Path(json.loads(stdout)["output"]), raw, stdout)
    assert result["verdict"] == "REJECTED"


def test_in_process_main_and_real_cli(tmp_path, capsys):
    assert M.main(["--output-dir", str(tmp_path / "demonstration")]) == 0
    printed = capsys.readouterr().out.encode("ascii")
    folder = Path(json.loads(printed)["output"])
    result = check_bundle(folder, M.encoded({"mode": "built_in_demo", "request": M.demo_request()}), printed)
    assert result["verdict"] == "DEMONSTRATED"
    task = request()
    document = M.submission(task, M.answer_for(task))
    raw = b" \n" + json.dumps(document, indent=1).encode("ascii") + b"\n"
    source = tmp_path / "request.json"
    source.write_bytes(raw)
    for mismatch in (False, True):
        if mismatch:
            document["proposal"]["request_sha256"] = "0" * 64
            raw = M.encoded(document)
            source.write_bytes(raw)
        completed = subprocess.run([sys.executable, "-B", str(RUNNER), "--input", str(source), "--output-dir", str(tmp_path / "cli")], capture_output=True, check=False)
        assert completed.returncode == int(mismatch) and completed.stderr == b""
        folder = Path(json.loads(completed.stdout)["output"])
        report = check_bundle(folder, raw, completed.stdout)
        assert report["verdict"] == ("MISMATCH" if mismatch else "MATCH")
    assert M.main(["--input", str(tmp_path / "missing"), "--output-dir", str(tmp_path / "missing_run")]) == 2
    assert json.loads(capsys.readouterr().out)["status"] == "REFUSED"

def test_semantic_mutations():
    source = RUNNER.read_text(encoding="utf-8")
    mutations = [
        ("allow_zero_weight", "([a-z])([1-9])", "([a-z])([0-9])", lambda m: assert_rejected(request("0>1w0", cutoff=1, degree=0, x=["1", "0"]), m)),
        ("ignore_weight", "int(digit) if digit is not None else 1", "1", lambda m: assert_triangle(m)),
        ("ignore_duplicate", "require(edge not in edges,", "require(True,", lambda m: assert_rejected(request("0>1<>1>0", cutoff=1, degree=0, x=["1", "0"]), m)),
        ("ignore_self_loop", "require(a != b,", "require(True,", lambda m: assert_self_loop(m)),
        ("strip_whitespace", 'tokens = text.split("<>")', 'tokens = text.strip().split("<>")', lambda m: assert_rejected(request("0 ", cutoff=0, degree=0, x=["1"]), m)),
        ("ignore_isolation", "require(not singles.intersection(v for edge in edges for v in edge),", "require(True,", lambda m: assert_rejected(request("0<>0>1", cutoff=1, degree=0, x=["1", "0"]), m)),
        ("ignore_truncation", '0 <= n <= 4 and graph["cutoff"] >= n', "0 <= n <= 4", lambda m: assert_rejected(request(cutoff=1), m)),
        ("no_orientation", "(-1) ** inversions * scale", "scale", lambda m: assert_triangle(m)),
        ("inverse_weight_ratio", "Fraction(face_weight(other, weights), face_weight(face, weights))", "Fraction(face_weight(face, weights), face_weight(other, weights))", lambda m: assert_triangle(m)),
        ("missing_internal_edge", "product *= weights[edge]", "product *= 1", lambda m: assert_triangle(m)),
        ("wrong_square_sign", "(-1) ** ((k + 1) * (n - k))", "(-1) ** (k * (n - k))", lambda m: assert_triangle(m)),
        ("drop_denominator", "Fraction(square, value.denominator)", "Fraction(square)", lambda m: assert_half(m)),
        ("drop_radical", "radicand *= prime", "radicand *= 1", lambda m: assert_half(m)),
        ("skip_binding", "matches = binding and encoded", "matches = True and encoded", lambda m: assert_binding(m)),
        ("partial_answer_compare", 'encoded(proposal["answer"]) == encoded(canonical)', 'proposal["answer"]["task_id"] == canonical["task_id"]', lambda m: assert_answer(m)),
        ("alias_request", "document = clone(document)", "document = document", lambda m: assert_alias(m)),
        ("skip_core_pin", "require(sha(raw) == CORE_SHA256,", "require(True,", lambda m: assert_pin(m)),
    ]
    killed, records = [], []
    for name, old, new, check in mutations:
        assert source.count(old) == 1, name
        baseline = ModuleType("baseline_" + name)
        baseline.__file__ = str(RUNNER)
        exec(compile(source, "<hodge_mutation_baseline>", "exec"), baseline.__dict__)
        check(baseline)
        mutant_source = source.replace(old, new, 1)
        compiled = compile(mutant_source, f"<hodge_mutant_{name}>", "exec")
        mutant = ModuleType("mutant_" + name)
        mutant.__file__ = str(RUNNER)
        exec(compiled, mutant.__dict__)
        try:
            check(mutant)
        except (AssertionError, ValueError, ZeroDivisionError, pytest.fail.Exception) as error:
            killed.append(name)
            records.append({"name": name, "source_sha256": M.sha(RUNNER.read_bytes()),
                            "mutant_sha256": M.sha(mutant_source.encode("utf-8")), "old": old, "new": new,
                            "compiled": True, "baseline_passed": True, "killed": True,
                            "failure_class": type(error).__name__, "reason": str(error)})
    assert len(killed) == len(mutations), sorted(set(row[0] for row in mutations) - set(killed))
    destination = os.environ.get("HODGE_MUTATION_REPORT")
    if destination:
        Path(destination).write_text(json.dumps({"count": len(records), "mutations": records}, indent=2), encoding="utf-8")
