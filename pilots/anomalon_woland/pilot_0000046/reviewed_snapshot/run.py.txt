CANDIDATE_ID = "hodge_companion_candidate"
CORE_RELATIVE = Path("omega_engine_education/documentation/030_the_symbol_game_proposal_fidelity_protocol_pfp/pilots/pilot_000017/run.py")
CORE_SHA256 = "256fe853bfa391b42ad3ebbff41501752fb2568bffdc746b470acd07e5638073"
MAX_BYTES = 131072
OUTPUT_ROOT = Path(__file__).resolve().parent / "output"


class InvalidInput(ValueError):
    pass


def require(condition, message):
    if not condition:
        raise InvalidInput(message)


def fields(value, names):
    require(type(value) is dict and set(value) == set(names), "Exact object fields required")


def encoded(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True, allow_nan=False).encode("ascii")


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def clone(value):
    return json.loads(encoded(value))


@lru_cache(maxsize=1)
def get_core():
    for root in Path(__file__).resolve().parents:
        path = root / CORE_RELATIVE
        if path.is_file():
            raw = path.read_bytes()
            require(sha(raw) == CORE_SHA256, "Pinned unweighted core changed")
            module = ModuleType("pinned_pilot_000017")
            module.__file__ = str(path)
            exec(compile(raw, str(path), "exec"), module.__dict__)
            return module
    raise InvalidInput("Pinned unweighted core unavailable")


def parse_graph(text, cutoff):
    require(type(text) is str and 0 < len(text) <= 256, "Nonempty bounded ASCII graph required")
    require(type(cutoff) is int and 0 <= cutoff <= 4, "Cutoff must be integer zero through four")
    tokens = text.split("<>")
    singles, vertices, edges, spellings = set(), set(), {}, []
    for token in tokens:
        match = re.fullmatch(r"([0-9a-z])(?:([><])([0-9a-z])(?:([a-z])([1-9]))?)?", token, flags=re.ASCII)
        require(match is not None, "Malformed graph token; no whitespace or empty segments")
        a, direction, b, marker, digit = match.groups()
        vertices.add(a)
        if direction is None:
            require(a not in singles, "Duplicate isolated vertex")
            singles.add(a)
            spellings.append({"token": token, "isolated": a})
            continue
        require(a != b, "Self-loop refused")
        vertices.add(b)
        edge = tuple(sorted((a, b)))
        require(edge not in edges, "Repeated or conflicting undirected edge refused")
        weight = int(digit) if digit is not None else 1
        edges[edge] = weight
        spellings.append({"token": token, "source": a if direction == ">" else b,
                          "target": b if direction == ">" else a, "weight_marker": marker,
                          "weight": weight})
    require(len(vertices) <= 6, "At most six vertices")
    require(not singles.intersection(v for edge in edges for v in edge), "An isolated token cannot also be incident to an edge")
    ordered = sorted(vertices)
    weighted = [{"edge": list(edge), "weight": edges[edge]} for edge in sorted(edges)]
    carrier = {"mode": "cliques", "vertices": ordered, "edges": [row["edge"] for row in weighted],
               "max_dimension": cutoff}
    return {"vertices": ordered, "weighted_edges": weighted, "spellings": spellings,
            "cutoff": cutoff, "canonical_carrier": carrier,
            "direction_role": "retained_spelling_only_not_boundary_orientation"}


def square_root(value):
    """Return q,d with sqrt(value)=q*sqrt(d); value has only primes 2,3,5,7."""
    require(type(value) is Fraction and value > 0, "Positive rational weight ratio required")
    remaining = value.numerator * value.denominator
    square, radicand = 1, 1
    for prime in (2, 3, 5, 7):
        exponent = 0
        while remaining % prime == 0:
            remaining //= prime
            exponent += 1
        square *= prime ** (exponent // 2)
        if exponent % 2:
            radicand *= prime
    require(remaining == 1, "Weight prime outside the declared parser profile")
    return Fraction(square, value.denominator), radicand


def radical_record(value):
    coefficient, radicand = value
    return {"coefficient": str(coefficient), "radicand": radicand if coefficient else 1}


def multiply_radical(value, scalar, radicand):
    coefficient, old_radical = value
    root_coefficient, root_radical = square_root(Fraction(old_radical * radicand))
    result = coefficient * scalar * root_coefficient
    return result, root_radical if result else 1


def face_weight(face, weights):
    product = 1
    for edge in combinations(face, 2):
        product *= weights[edge]
    return product


def apply_complement(vertices, weights, k, values):
    """Internal exact map on radical monomials; validated callers supply a full simplex."""
    n = len(vertices) - 1
    source = tuple(combinations(vertices, k + 1))
    target_degree = n - k - 1
    target = tuple(combinations(vertices, target_degree + 1))
    positions = {face: i for i, face in enumerate(target)}
    output = [(Fraction(0), 1)] * len(target)
    require(len(values) == len(source), "Complement vector size mismatch")
    for face, value in zip(source, values, strict=True):
        other = tuple(v for v in vertices if v not in face)
        sequence = face + other
        inversions = sum(sequence[i] > sequence[j] for i in range(n + 1) for j in range(i + 1, n + 1))
        scale, radical = square_root(Fraction(face_weight(other, weights), face_weight(face, weights)))
        output[positions[other]] = multiply_radical(value, (-1) ** inversions * scale, radical)
    return target_degree, target, tuple(output)


def weighted_complement(graph, k, cochain):
    core = get_core()
    vertices = tuple(graph["vertices"])
    n = len(vertices) - 1
    weights = {tuple(row["edge"]): row["weight"] for row in graph["weighted_edges"]}
    require(0 <= n <= 4 and graph["cutoff"] >= n, "Complement requires the entire simplex of dimension zero through four")
    require(set(weights) == set(combinations(vertices, 2)), "Complement requires every simplex edge")
    require(type(k) is int and -1 <= k <= n, "Complement degree outside augmented simplex")
    source = tuple(combinations(vertices, k + 1))
    x = core.vector(cochain, len(source))
    target_degree, target, output = apply_complement(vertices, weights, k, tuple((q, 1) for q in x))
    _, _, twice = apply_complement(vertices, weights, target_degree, output)
    sign = (-1) ** ((k + 1) * (n - k))
    expected = tuple((sign * q, 1) for q in x)
    require(twice == expected, "Exact complement square check failed")
    return {"map": "weighted_augmented_complement", "simplex_dimension": n, "degree": k,
            "source_basis": [list(face) for face in source], "target_degree": target_degree,
            "target_basis": [list(face) for face in target],
            "source_face_weights": [face_weight(face, weights) for face in source],
            "target_face_weights": [face_weight(face, weights) for face in target],
            "value": [radical_record(v) for v in output], "twice": [radical_record(v) for v in twice],
            "double_sign": sign, "square_identity_checked": True, "geometric_hodge_star_claim": False}


def answer_for(request):
    core = get_core()
    core.json_value(request)
    require(len(encoded(request)) <= MAX_BYTES, "Request byte limit")
    fields(request, ("task_id", "kind", "graph", "cutoff", "degree", "cochain"))
    require(type(request["task_id"]) is str and re.fullmatch(r"[A-Za-z0-9_.-]{1,32}", request["task_id"]), "Bounded task identifier required")
    require(request["kind"] in ("unweighted", "complement"), "Only unweighted analysis or algebraic complement is supported")
    graph = parse_graph(request["graph"], request["cutoff"])
    if request["kind"] == "unweighted":
        query = {"case_id": request["task_id"], "kind": "analyse", "complex": graph["canonical_carrier"],
                 "degree": request["degree"], "cochain": request["cochain"]}
        data = core.answer_for(query)
    else:
        data = weighted_complement(graph, request["degree"], request["cochain"])
    return clone({"task_id": request["task_id"], "kind": request["kind"], "graph": graph, "data": data,
                  "unweighted_core_sha256": CORE_SHA256, "weighted_laplacian_claim": False,
                  "live_model_claim": False, "authority_effect": "None", "genesis_effect": "None", "realizes": None})


def submission(request, answer):
    return clone({"request": request, "proposal": {"request_sha256": sha(encoded(request)), "answer": answer}})


def evaluate(document):
    core = get_core()
    core.json_value(document)
    require(len(encoded(document)) <= MAX_BYTES, "Document byte limit")
    document = clone(document)
    fields(document, ("request", "proposal"))
    fields(document["proposal"], ("request_sha256", "answer"))
    canonical = answer_for(document["request"])
    proposal = document["proposal"]
    binding = type(proposal["request_sha256"]) is str and proposal["request_sha256"] == sha(encoded(document["request"]))
    matches = binding and encoded(proposal["answer"]) == encoded(canonical)
    return {"verdict": "MATCH" if matches else "MISMATCH", "binding_ok": binding,
            "request": document["request"], "proposal": proposal, "canonical": canonical}


def demo_request():
    return {"task_id": "weighted_triangle", "kind": "complement", "graph": "0>1w4<>0>2w9<>1>2w1",
            "cutoff": 2, "degree": 1, "cochain": ["1", "2", "3"]}


def capture_input(path, target):
    """Copy the full readable input without truncation; evaluation has a separate size cap."""
    hasher, size = hashlib.sha256(), 0
    with target.open("xb") as destination:
        if path is None:
            raw = encoded({"mode": "built_in_demo", "request": demo_request()})
            destination.write(raw)
            hasher.update(raw)
            size = len(raw)
        else:
            with path.open("rb") as source:
                while chunk := source.read(65536):
                    destination.write(chunk)
                    hasher.update(chunk)
                    size += len(chunk)
    return size, hasher.hexdigest()


def run_file(input_path, output_root):
    folder = output_root / uuid4().hex
    folder.mkdir(parents=True, exist_ok=False)
    size, input_hash = capture_input(input_path, folder / "input.bin")
    try:
        require(0 < size <= MAX_BYTES, "Evaluation input byte limit")
        core = get_core()
        if input_path is None:
            report = {"verdict": "DEMONSTRATED", "answer": answer_for(demo_request())}
        else:
            report = evaluate(core.decode((folder / "input.bin").read_bytes()))
        status = report["verdict"]
    except (ValueError, TypeError, RecursionError) as error:
        status, report = "REJECTED", {"verdict": "REJECTED", "reason": str(error)}
    stdout = encoded({"status": status, "output": str(folder.resolve())}) + b"\n"
    artifacts = {"result.json": encoded(report), "stdout.txt": stdout, "stderr.txt": b""}
    for name, raw in artifacts.items():
        (folder / name).write_bytes(raw)
    receipt = {"candidate_id": CANDIDATE_ID, "status": status, "input_bytes": size,
               "input_sha256": input_hash, "input_complete": True,
               "code_sha256": sha(Path(__file__).read_bytes()), "core_sha256": CORE_SHA256,
               "model_calls": 0, "cost_usd": "0", "authority_effect": "None",
               "artifact_sha256": {"input.bin": input_hash, **{name: sha(raw) for name, raw in artifacts.items()}}}
    (folder / "receipt.json").write_bytes(encoded(receipt))
    return status, stdout


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", type=Path)
    parser.add_argument("--output-dir", type=Path, default=OUTPUT_ROOT)
    args = parser.parse_args(argv)
    try:
        status, stdout = run_file(args.input, args.output_dir)
    except (OSError, ValueError) as error:
        print(json.dumps({"status": "REFUSED", "reason": str(error)}, ensure_ascii=True))
        return 2
    sys.stdout.buffer.write(stdout)
    return 1 if status in ("MISMATCH", "REJECTED") else 0


if __name__ == "__main__":
    sys.exit(main())
