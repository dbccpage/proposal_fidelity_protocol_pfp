"""Pytest test suite for Demo 4 Research Dossier Pair."""

from __future__ import annotations

import importlib.util
import json
import sys
import unittest
from pathlib import Path
from typing import Any

import yaml  # type: ignore

DEMO4_DIR = Path(__file__).parent
DOSSIER1_DIR = DEMO4_DIR / "dossier_1_f2_spin_triangle"
DOSSIER2_DIR = DEMO4_DIR / "dossier_2_exact_lp_certificate"


def _load_module(name: str, path: Path) -> Any:
    spec = importlib.util.spec_from_file_location(name, path)
    if spec is None or spec.loader is None:
        raise ImportError(f"Cannot load module {name} from {path}")
    mod = importlib.util.module_from_spec(spec)
    sys.modules[name] = mod
    spec.loader.exec_module(mod)
    return mod


d1_checker = _load_module("d1_checker", DOSSIER1_DIR / "checker.py")
d2_checker = _load_module("d2_checker", DOSSIER2_DIR / "checker.py")
run_demo4_mod = _load_module("run_demo4_mod", DEMO4_DIR / "run_demo4.py")


class TestDemo4ResearchDossiers(unittest.TestCase):
    def setUp(self) -> None:
        with open(DOSSIER1_DIR / "input.yaml", "r", encoding="utf-8") as f:
            self.d1_input = yaml.safe_load(f)
        with open(DOSSIER1_DIR / "witness_valid.json", "r", encoding="utf-8") as f:
            self.d1_witness_valid = json.load(f)
        with open(DOSSIER1_DIR / "witness_near_miss.json", "r", encoding="utf-8") as f:
            self.d1_witness_near = json.load(f)

        with open(DOSSIER2_DIR / "input.yaml", "r", encoding="utf-8") as f:
            self.d2_input = yaml.safe_load(f)
        with open(DOSSIER2_DIR / "witness_valid.json", "r", encoding="utf-8") as f:
            self.d2_witness_valid = json.load(f)
        with open(DOSSIER2_DIR / "witness_near_miss.json", "r", encoding="utf-8") as f:
            self.d2_witness_near = json.load(f)

    # --------------------------------------------------------------------------
    # Dossier 1 Tests: F2 Frustrated Spin Triangle
    # --------------------------------------------------------------------------

    def test_dossier1_nilpotence_identity(self) -> None:
        d0 = self.d1_input["matrices"]["d0"]
        d1 = self.d1_input["matrices"]["d1"]
        d1_d0 = d1_checker.matmul_mod2(d1, d0)
        self.assertTrue(all(all(x == 0 for x in row) for row in d1_d0))

    def test_dossier1_curl_obstruction(self) -> None:
        d1 = self.d1_input["matrices"]["d1"]
        c = self.d1_input["target_cochain"]["values"]
        curl = d1_checker.matvec_mod2(d1, c)[0]
        self.assertEqual(curl, 1)

    def test_dossier1_valid_witness_certified(self) -> None:
        receipt = d1_checker.verify_dossier1(self.d1_input, self.d1_witness_valid)
        self.assertEqual(receipt.status, "CERTIFIED_OBSTRUCTION")
        self.assertTrue(receipt.admitted)
        self.assertTrue(receipt.nilpotence_verified)
        self.assertEqual(receipt.curl_d1_c, 1)
        self.assertEqual(receipt.actual_defect_weight, 1)
        self.assertTrue(receipt.is_provably_minimal)

    def test_dossier1_near_miss_refused(self) -> None:
        receipt = d1_checker.verify_dossier1(self.d1_input, self.d1_witness_near)
        self.assertEqual(receipt.status, "REFUSED_FALSE_ZERO_DEFECT_CLAIM")
        self.assertFalse(receipt.admitted)
        self.assertEqual(receipt.actual_defect_weight, 1)
        self.assertEqual(receipt.claimed_defect_weight, 0)

    # --------------------------------------------------------------------------
    # Dossier 2 Tests: Exact Rational LP Certificate
    # --------------------------------------------------------------------------

    def test_dossier2_valid_witness_certified(self) -> None:
        receipt = d2_checker.verify_lp_certificate(
            self.d2_input, self.d2_witness_valid, case_name="valid_test"
        )
        self.assertEqual(receipt.status, "CERTIFIED_OPTIMAL_LP_GAP_ZERO")
        self.assertTrue(receipt.admitted)
        self.assertTrue(receipt.primal_feasible)
        self.assertTrue(receipt.dual_feasible)
        self.assertTrue(receipt.is_exact_zero_gap)
        self.assertEqual(receipt.primal_objective, "1")
        self.assertEqual(receipt.dual_objective, "1")
        self.assertEqual(receipt.duality_gap, "0")

    def test_dossier2_near_miss_float_refused(self) -> None:
        float_case = self.d2_witness_near["test_cases"][0]
        receipt = d2_checker.verify_lp_certificate(
            self.d2_input, float_case, case_name="float_test"
        )
        self.assertEqual(receipt.status, "REFUSED_PRIMAL_INFEASIBLE")
        self.assertFalse(receipt.admitted)
        self.assertFalse(receipt.is_exact_zero_gap)

    def test_dossier2_near_miss_infeasible_dual_refused(self) -> None:
        dual_infeasible_case = self.d2_witness_near["test_cases"][1]
        receipt = d2_checker.verify_lp_certificate(
            self.d2_input, dual_infeasible_case, case_name="dual_infeasible_test"
        )
        self.assertEqual(receipt.status, "REFUSED_DUAL_INFEASIBLE")
        self.assertFalse(receipt.admitted)
        self.assertFalse(receipt.dual_feasible)

    # --------------------------------------------------------------------------
    # Full Integration Runner Test
    # --------------------------------------------------------------------------

    def test_run_demo4_integration(self) -> None:
        summary = run_demo4_mod.run_demo4()
        self.assertEqual(summary["dossier_1_status"], "CERTIFIED_OBSTRUCTION")
        self.assertEqual(summary["dossier_2_status"], "CERTIFIED_OPTIMAL_LP_GAP_ZERO")


if __name__ == "__main__":
    unittest.main()
